### 2.1 依赖注入系统 (P0 - 紧急)

#### 问题分析

当前 `DiContainer` 是一个**占位实现**，完全没有实际功能：

```rust
// src/di/container.rs - 当前实现
pub struct DiContainer;

impl DiContainer {
    pub fn new() -> Self {
        Self
    }
}
```

这导致：
1. 组件依赖关系无法自动管理
2. 测试时无法轻松 Mock 依赖
3. 违反了 DDD 中依赖反转原则的实现要求

#### 优化建议

**P0-1: 实现完整的 DI 容器**

```rust
use std::any::{Any, TypeId};
use std::collections::HashMap;
use std::sync::{Arc, RwLock};

pub struct DiContainer {
    services: RwLock<HashMap<TypeId, Arc<dyn Any + Send + Sync>>>,
    factories: RwLock<HashMap<TypeId, Box<dyn Fn() -> Arc<dyn Any + Send + Sync> + Send + Sync>>>,
}

impl DiContainer {
    pub fn new() -> Self {
        Self {
            services: RwLock::new(HashMap::new()),
            factories: RwLock::new(HashMap::new()),
        }
    }

    /// 注册单例服务
    pub fn register_singleton<T: 'static + Send + Sync>(&self, instance: Arc<T>) {
        let mut services = self.services.write().unwrap();
        services.insert(TypeId::of::<T>(), instance);
    }

    /// 注册工厂方法（每次创建新实例）
    pub fn register_factory<T, F>(&self, factory: F)
    where
        T: 'static + Send + Sync,
        F: Fn() -> Arc<T> + 'static + Send + Sync,
    {
        let mut factories = self.factories.write().unwrap();
        factories.insert(
            TypeId::of::<T>(),
            Box::new(move || factory() as Arc<dyn Any + Send + Sync>),
        );
    }

    /// 解析服务
    pub fn resolve<T: 'static + Send + Sync>(&self) -> Option<Arc<T>> {
        // 先查单例
        {
            let services = self.services.read().unwrap();
            if let Some(service) = services.get(&TypeId::of::<T>()) {
                return service.clone().downcast::<T>().ok();
            }
        }

        // 再查工厂
        let factories = self.factories.read().unwrap();
        if let Some(factory) = factories.get(&TypeId::of::<T>()) {
            let instance = factory();
            instance.downcast::<T>().ok()
        } else {
            None
        }
    }
}
```

**P2-4: 工具组合优化**

```rust
impl ComposedTool {
    pub async fn execute(&self, input: ToolInput, ctx: ExecutionContext) -> Result<ToolOutput> {
        match &self.composition_type {
            CompositionType::Chain(tools) => {
                let mut current_input = input;
                for tool_id in tools {
                    let tool = self.registry.get(tool_id)?;
                    let output = tool.execute(current_input, ctx.clone()).await?;
                    current_input = ToolInput::new(output.result);
                }
                Ok(current_input.into())
            }
            CompositionType::Parallel(tools) => {
                let futures: Vec<_> = tools.iter()
                    .map(|id| {
                        let tool = self.registry.get(id);
                        let input = input.clone();
                        let ctx = ctx.clone();
                        async move { tool?.execute(input, ctx).await }
                    })
                    .collect();
                
                let results = join_all(futures).await;
                // 合并结果
                self.merge_results(results)
            }
            CompositionType::Conditional { condition, then_tool, else_tool } => {
                if self.evaluate_condition(&condition, &input.params)? {
                    self.registry.get(then_tool)?.execute(input, ctx).await
                } else if let Some(else_id) = else_tool {
                    self.registry.get(else_id)?.execute(input, ctx).await
                } else {
                    Ok(ToolOutput::success(Value::Null))
                }
            }
        }
    }
}
```

```rust
    failure_count: AtomicU32,

    state_change_count: AtomicU32,
}

impl CircuitBreakerMetrics {
    pub fn record_success(&self) {
        self.success_count.fetch_add(1, Ordering::Relaxed);
        self.consecutive_failures.store(0, Ordering::Relaxed);  // 重置连续失败

    pub fn record_failure(&self) {
        self.failure_count.fetch_add(1, Ordering::Relaxed);

    }

        self.consecutive_failures.load(Ordering::Relaxed)
```

**P1-6: 实现完整的工作流控制**

```rust
pub enum FlowNode {
    Empty,
    Tool { id: String, tool_name: String, params: Value },
    Chain(Vec<FlowNode>),
    Parallel(Vec<FlowNode>),
    Switch {
        id: String,
        condition: String,
        cases: HashMap<String, FlowNode>,
        default: Option<Box<FlowNode>>,
    },
    Loop {
        id: String,
        condition: String,
        body: Box<FlowNode>,
        max_iterations: u32,
    },
}

impl RefactoredWorkflowEngine {
    async fn execute_switch(
        &self,
        switch: &SwitchNode,
        context: &mut DataContext,
        tracker: Arc<ExecutionTracker>,
    ) -> Result<()> {
        let condition_result = self.evaluate_expression(&switch.condition, context)?;
        
        let branch = switch.cases.get(&condition_result.to_string())
            .or(switch.default.as_ref())
            .ok_or_else(|| WorkflowError::execution("Switch 无匹配分支"))?;
        
        self.execute_flow(branch, context, tracker).await
    }
    
    async fn execute_loop(
        &self,
        loop_node: &LoopNode,
        context: &mut DataContext,
        tracker: Arc<ExecutionTracker>,
    ) -> Result<()> {
        let mut iterations = 0;
        
        while iterations < loop_node.max_iterations {
            if !self.evaluate_condition(&loop_node.condition, context)? {
                break;
            }
            
            self.execute_flow(&loop_node.body, context, tracker.clone()).await?;
            iterations += 1;
        }
        
        Ok(())
    }
}
```

**P1-7: 实现背压机制**

```rust
pub struct BackpressureController {
    max_queue_size: usize,
    current_size: AtomicUsize,
    drop_strategy: DropStrategy,
}

pub enum DropStrategy {
    DropOldest,
    DropNewest,
    Block,
}

impl BackpressureController {
    pub async fn acquire(&self) -> Result<BackpressurePermit> {
        loop {
            let current = self.current_size.load(Ordering::Relaxed);
            
            if current < self.max_queue_size {
                if self.current_size.compare_exchange(
                    current,
                    current + 1,
                    Ordering::AcqRel,
                    Ordering::Relaxed,
                ).is_ok() {
                    return Ok(BackpressurePermit {
                        controller: self,
                    });
                }
            }
            
            match self.drop_strategy {
                DropStrategy::Block => {
                    tokio::time::sleep(Duration::from_millis(10)).await;
                }
                DropStrategy::DropNewest => {
                    return Err(WorkflowError::backpressure("队列已满，拒绝新请求"));
                }
                DropStrategy::DropOldest => {
                    // 实现丢弃最旧任务的逻辑
                }
            }
        }
    }
}
```

**P2-6: 优化并行执行**

```rust
impl RefactoredWorkflowEngine {
    async fn execute_nodes_parallel_optimized(
        &self,
        ctx: &WorkflowLoopContext<'_>,
        node_ids: &[String],
        context: &mut DataContext,
        scheduler: &mut DagScheduler,
    ) -> Vec<Result<ComponentOutput>> {
        // 使用 JoinSet 而非 join_all，支持取消
        let mut join_set = JoinSet::new();
        
        for node_id in node_ids {
            let node_id = node_id.clone();
            let permit = self.semaphore.clone().acquire_owned().await.ok();
            let executor = self.executor.clone();
            let registry = ctx.component_registry.clone();
            let workflow_id = ctx.workflow_id;
            
            join_set.spawn(async move {
                let _permit = permit;
                let component = registry.get(&node_id)?;
                let exec_ctx = ExecutionContext::new().with_workflow_id(workflow_id);
                executor.execute(component.as_ref(), &mut context, &exec_ctx).await
            });
        }
        
        // 收集结果，支持取消
        let mut results = Vec::with_capacity(node_ids.len());
        while let Some(result) = join_set.join_next().await {
            results.push(result?);
        }
        
        results
    }
}
```

```rust
pub trait DomainEvent: Send + Sync + Clone + 'static {
    fn event_type(&self) -> &'static str;
    fn occurred_at(&self) -> DateTime<Utc>;
    fn aggregate_id(&self) -> Uuid;
}

pub struct DomainEventBus {
    handlers: DashMap<String, Vec<Box<dyn EventHandler>>>,
    event_store: Arc<dyn EventStore>,
}

impl DomainEventBus {
    pub async fn publish<E: DomainEvent>(&self, event: E) -> Result<()> {
        // 存储事件
        self.event_store.append(&event).await?;
        
        // 通知订阅者
        if let Some(handlers) = self.handlers.get(event.event_type()) {
            for handler in handlers.iter() {
                handler.handle(&event).await?;
            }
        }
        
        Ok(())
    }
    
    pub fn subscribe<E: DomainEvent, H: EventHandler<E> + 'static>(&mut self, handler: H) {
        self.handlers
            .entry(E::event_type_static().to_string())
            .or_default()
            .push(Box::new(handler));
    }
}
```

**P1-9: 实现 Unit of Work**

```rust
pub struct UnitOfWork {
    transaction: Option<Transaction>,
    event_bus: Arc<DomainEventBus>,
    pending_events: Vec<Box<dyn DomainEvent>>,
}

impl UnitOfWork {
    pub async fn begin(&mut self) -> Result<()> {
        self.transaction = Some(Transaction::begin().await?);
        Ok(())
    }
    
    pub fn add_event<E: DomainEvent>(&mut self, event: E) {
        self.pending_events.push(Box::new(event));
    }
    
    pub async fn commit(mut self) -> Result<()> {
        // 提交事务
        if let Some(tx) = self.transaction.take() {
            tx.commit().await?;
        }
        
        // 发布领域事件
        for event in self.pending_events.drain(..) {
            self.event_bus.publish(event).await?;
        }
        
        Ok(())
    }
    
    pub async fn rollback(mut self) -> Result<()> {
        if let Some(tx) = self.transaction.take() {
            tx.rollback().await?;
        }
        self.pending_events.clear();
        Ok(())
    }
}
```

```rust
    pub fn histogram(&self, name: &str) -> Histogram {
        self.histograms.get(name).cloned().unwrap_or_default()
    }
    
    /// 导出 Prometheus 格式
    pub fn export_prometheus(&self) -> String {
        let mut output = String::new();
        
        for (name, counter) in self.counters.iter() {
            output.push_str(&format!("{} {}\n", name, counter.load(Ordering::Relaxed)));
        }
        
        for (name, gauge) in self.gauges.iter() {
            output.push_str(&format!("{} {}\n", name, gauge.load(Ordering::Relaxed)));
        }
        
        output
    }
}
```
